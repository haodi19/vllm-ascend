cmake_minimum_required(VERSION 3.16)
project(host_sync_device LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUN_MODE "npu" CACHE STRING "Kernel execution mode")
set(SOC_VERSION "ascend950" CACHE STRING "Target SoC")
set(ASCEND_COMPUTE_UNIT "${SOC_VERSION}" CACHE STRING "Ascend compute unit")
if(NOT SOC_VERSION MATCHES "^ascend950")
    message(FATAL_ERROR "Stage 2 currently targets Ascend950 only")
endif()
if(NOT ASCEND_CANN_PACKAGE_PATH)
    message(FATAL_ERROR "Pass -DASCEND_CANN_PACKAGE_PATH=<CANN root>")
endif()
foreach(relative_path
        tools/tikcpp/ascendc_kernel_cmake
        compiler/tikcpp/ascendc_kernel_cmake
        ascendc_devkit/tikcpp/samples/cmake)
    if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/${relative_path}/ascendc.cmake")
        set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/${relative_path}")
        break()
    endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "Cannot find ascendc.cmake under the selected CANN root")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
ascendc_library(host_sync_device SHARED handshake.cpp)
set_target_properties(host_sync_device PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
